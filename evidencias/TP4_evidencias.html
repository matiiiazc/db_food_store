       c.nombre AS categoria, sum(dp.subtotal) AS facturado

Parte 3 — Ranking con función de ventana (spec a)

Especificación precisa

"Para cada cliente vigente con al menos un pedido: nombre completo, total gastado
 (suma de detalle_pedido.subtotal) y puesto en ranking de mayor a menor gasto,
 sin colapsar filas. Empates comparten el mismo puesto. No usés SELECT *."

Versión 1 — RANK() (ventana)

-- 3a
WITH g AS (
    SELECT p.cliente_id, sum(dp.subtotal) AS total
    FROM pedido p JOIN detalle_pedido dp ON dp.pedido_id = p.id
    GROUP BY p.cliente_id
)
SELECT cl.id, cl.nombre, cl.apellido, g.total,
       RANK() OVER (ORDER BY g.total DESC) AS puesto
FROM g JOIN cliente cl ON cl.id = g.cliente_id
ORDER BY puesto, cl.id;

Versión 2 — subconsulta correlacionada replicando RANK

-- 3b

SELECT cl.id, cl.nombre, cl.apellido, g.total,
       (SELECT count(*) + 1 FROM g h WHERE h.total > g.total)::bigint AS puesto
FROM g JOIN cliente cl ON cl.id = g.cliente_id
ORDER BY (SELECT count(*) + 1 FROM g h WHERE h.total > g.total), cl.id;

Parte 3 — Subconsulta correlacionada (spec b)

Especificación precisa

"Para cada producto: nombre, precio y la diferencia entre su precio y el precio
 promedio de los productos de su misma categoría (subconsulta correlacionada).
 Incluí todos los productos. No usés SELECT *."

Versión 1 — subconsulta correlacionada

-- 3e
SELECT p.id, p.nombre AS producto, p.precio,
       round(p.precio - (SELECT avg(p2.precio) FROM producto p2
                         WHERE p2.categoria_id = p.categoria_id), 2) AS dif_vs_categoria
FROM producto p; -- (LIMIT 50 solo para el preview)

Versión 2 — join + agregación (estructura distinta)

-- 3f
SELECT p.id, p.nombre AS producto, p.precio,
       round(p.precio - c.promedio, 2) AS dif_vs_categoria
FROM producto p
JOIN (SELECT categoria_id, avg(precio) AS promedio
      FROM producto GROUP BY categoria_id) c
  ON c.categoria_id = p.categoria_id; -- (LIMIT 50 solo para el preview)
